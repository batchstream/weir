package search

import (
    "fmt"
    "runtime"
    "strings"
    "testing"

    pb "github.com/batchstream/weir-protocol/api/weir/v1"
    "github.com/batchstream/weir/internal/execution"
    "github.com/batchstream/weir/internal/testutil"
)

func reviewHeaderInput() []byte {
    var builder strings.Builder
    builder.WriteString("GET / HTTP/1.1\r\nHost: localhost\r\n")
    for i := 0; i < 100000; i++ {
        fmt.Fprintf(&builder, "X-%06d: x\r\n", i)
    }
    builder.WriteString("\r\n")
    raw := []byte(builder.String())
    return raw
}

func TestReviewCorrectedNativePreparedInputOwnership(t *testing.T) {
    raw := reviewHeaderInput()
    document := &pb.Document{ContentType: "application/http", Data: raw}
    request := &pb.NativeRequest{Resource: "records", Request: document}
    command := testutil.NativeCommand(request)
    adapter := &Adapter{}
    runtime.GC()
    var before runtime.MemStats
    runtime.ReadMemStats(&before)
    var plans [3]*execution.Plan
    for i := range plans {
        work, failure := adapter.PrepareCommand(t.Context(), 1, command)
        if failure != nil { t.Fatal(failure) }
        plans[i] = work
    }
    runtime.GC()
    var after runtime.MemStats
    runtime.ReadMemStats(&after)
    retained := int64(after.HeapAlloc) - int64(before.HeapAlloc)
    native := plans[0].Backend.(*nativePlan)
    if retained > 256<<10 { t.Fatal("pre-admission parser ownership remains", retained) }
    if len(native.raw) != len(raw) || native.working < 8*len(raw) { t.Fatal("borrowed input or reserved parser bound missing") }
    probe := &nativeMemoryProbe{active:true, limit: plans[0].WorkingBytes-1}
    plans[0].NativeIO = probe
    capture := &nativeCapture{}
    var rejectedBefore, rejectedAfter runtime.MemStats
    runtime.ReadMemStats(&rejectedBefore)
    end := adapter.executeNative(t.Context(), plans[0], capture.Emit)
    runtime.ReadMemStats(&rejectedAfter)
    rejectedAllocated := rejectedAfter.TotalAlloc-rejectedBefore.TotalAlloc
    if end.Completion != pb.NativeCompletion_NATIVE_NOT_STARTED || end.GetFailure().GetCode() != pb.FailureCode_RESOURCE_EXHAUSTED || rejectedAllocated > 32<<10 { t.Fatal("parser ran before rejected reservation", end, rejectedAllocated) }
    t.Logf("raw_input=%d prepared3_retained_heap=%d routed_plan_input=%d request_bound=%d initial_working=%d rejected_execute_allocated=%d", len(raw), retained, plans[0].Bytes, native.working, nativeBudget+native.working, rejectedAllocated)
    runtime.KeepAlive(plans)
    runtime.KeepAlive(request)
    runtime.KeepAlive(command)
}
