package main

import (
	"context"
	"encoding/json"
	"fmt"
	"net/http"
	"os"
	"time"

	"go.mongodb.org/mongo-driver/v2/bson"
	"go.mongodb.org/mongo-driver/v2/mongo"
	"go.mongodb.org/mongo-driver/v2/mongo/options"
)

func main() {
	ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
	defer cancel()
	opts := options.Client().ApplyURI("mongodb://127.0.0.1:27028/?directConnection=true&serverMonitoringMode=poll").SetRetryReads(false).SetRetryWrites(false).SetMaxAdaptiveRetries(0).SetMaxPoolSize(2)
	client, err := mongo.Connect(opts)
	if err != nil {
		panic(err)
	}
	defer client.Disconnect(context.Background())
	filter := bson.D{{Key: "name", Value: "weir_test_86965_1"}}
	names, err := client.ListDatabaseNames(ctx, filter)
	if err != nil || len(names) != 0 {
		panic(fmt.Sprintf("owned Mongo fixture remains: %v %v", names, err))
	}
	request, err := http.NewRequestWithContext(ctx, http.MethodHead, "http://127.0.0.1:19200/weir_m2_86965_1", nil)
	if err != nil {
		panic(err)
	}
	transport := &http.Transport{Proxy: nil}
	observer := &http.Client{Transport: transport, Timeout: 5 * time.Second}
	defer transport.CloseIdleConnections()
	response, err := observer.Do(request)
	if err != nil {
		panic(err)
	}
	response.Body.Close()
	if response.StatusCode != http.StatusNotFound {
		panic(fmt.Sprintf("owned Search fixture remains: %d", response.StatusCode))
	}
	receipt := struct {
		ControllerPID int      `json:"controller_pid"`
		MongoDB       string   `json:"mongodb_owned_database"`
		SearchIndex   string   `json:"search_owned_index"`
		MongoAbsent   bool     `json:"mongodb_absent"`
		SearchStatus  int      `json:"search_head_status"`
		Cleanup       string   `json:"cleanup"`
		Errors        []string `json:"errors"`
	}{ControllerPID: 86965, MongoDB: "weir_test_86965_1", SearchIndex: "weir_m2_86965_1", MongoAbsent: true, SearchStatus: response.StatusCode, Cleanup: "controller testfixture cleanup dropped its uniquely owned database/index; independently observed absence after timing", Errors: []string{}}
	raw, err := json.MarshalIndent(receipt, "", "  ")
	if err != nil {
		panic(err)
	}
	if err := os.WriteFile("/tmp/weir-scan-diagnostic-vq_qyjmb/cleanup.json", append(raw, '\n'), 0644); err != nil {
		panic(err)
	}
	fmt.Println(string(raw))
}
