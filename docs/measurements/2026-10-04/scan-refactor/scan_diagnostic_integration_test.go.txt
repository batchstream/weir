const scanDiagnosticDocuments = 1024
const scanDiagnosticPayload = 1024

type scanDiagnosticCounts struct {
	mu        sync.Mutex
	Commands  map[string]int `json:"commands"`
	Histogram map[int]int    `json:"requested_batch_histogram"`
}

func newScanDiagnosticCounts() *scanDiagnosticCounts {
	counts := &scanDiagnosticCounts{Commands: make(map[string]int), Histogram: make(map[int]int)}
	return counts
}

func (c *scanDiagnosticCounts) record(name string, batch int) {
	c.mu.Lock()
	defer c.mu.Unlock()
	c.Commands[name]++
	if batch > 0 {
		c.Histogram[batch]++
	}
}

func (c *scanDiagnosticCounts) snapshot() *scanDiagnosticCounts {
	c.mu.Lock()
	defer c.mu.Unlock()
	copy := newScanDiagnosticCounts()
	for name, count := range c.Commands {
		copy.Commands[name] = count
	}
	for size, count := range c.Histogram {
		copy.Histogram[size] = count
	}
	return copy
}

func scanDiagnosticDifference(after, before *scanDiagnosticCounts) *scanDiagnosticCounts {
	counts := newScanDiagnosticCounts()
	for name, count := range after.Commands {
		if delta := count - before.Commands[name]; delta != 0 {
			counts.Commands[name] = delta
		}
	}
	for size, count := range after.Histogram {
		if delta := count - before.Histogram[size]; delta != 0 {
			counts.Histogram[size] = delta
		}
	}
	return counts
}

type scanDiagnosticRound struct {
	Variant    string                `json:"variant"`
	Source     string                `json:"source"`
	Backend    string                `json:"backend"`
	Order      string                `json:"order"`
	Round      int                   `json:"round"`
	Documents  int                   `json:"documents"`
	RPCs       int                   `json:"rpcs"`
	ElapsedNS  int64                 `json:"elapsed_ns"`
	DocsSecond float64               `json:"documents_per_second"`
	Counts     *scanDiagnosticCounts `json:"native_observation"`
}

func scanDiagnosticSearchProxy(t *testing.T, upstream string, counts *scanDiagnosticCounts) string {
	t.Helper()
	transport := &http.Transport{Proxy: nil, DisableCompression: true, MaxConnsPerHost: 4, MaxIdleConnsPerHost: 4}
	client := &http.Client{Transport: transport, Timeout: 8 * time.Second}
	t.Cleanup(transport.CloseIdleConnections)
	proxy := httptest.NewServer(http.HandlerFunc(func(writer http.ResponseWriter, inbound *http.Request) {
		body, err := io.ReadAll(io.LimitReader(inbound.Body, 1<<20))
		if err != nil {
			writer.WriteHeader(http.StatusBadGateway)
			return
		}
		name, size := inbound.Method+" "+inbound.URL.Path, 0
		if inbound.URL.Path == "/_search" {
			var query struct {
				Size int `json:"size"`
			}
			if json.Unmarshal(body, &query) != nil {
				writer.WriteHeader(http.StatusBadRequest)
				return
			}
			size = query.Size
		}
		counts.record(name, size)
		request, err := http.NewRequestWithContext(inbound.Context(), inbound.Method, upstream+inbound.URL.RequestURI(), bytes.NewReader(body))
		if err != nil {
			writer.WriteHeader(http.StatusBadGateway)
			return
		}
		request.Header = inbound.Header.Clone()
		request.GetBody = nil
		response, err := client.Do(request)
		if err != nil {
			writer.WriteHeader(http.StatusBadGateway)
			return
		}
		defer response.Body.Close()
		for name, values := range response.Header {
			for _, value := range values {
				writer.Header().Add(name, value)
			}
		}
		writer.WriteHeader(response.StatusCode)
		_, _ = io.Copy(writer, response.Body)
	}))
	t.Cleanup(proxy.Close)
	return proxy.URL
}

func scanDiagnosticTraverse(t *testing.T, client pb.StoreServiceClient, target, backend string) int {
	t.Helper()
	ctx, cancel := context.WithTimeout(t.Context(), 60*time.Second)
	defer cancel()
	seen := make(map[string]bool, scanDiagnosticDocuments)
	expectedPadding := strings.Repeat("x", scanDiagnosticPayload)
	var token []byte
	for rpc := 1; rpc <= 6; rpc++ {
		request := &pb.ScanRequest{Resource: target, PageSize: 256, ContinuationToken: bytes.Clone(token)}
		variant := &pb.Command_Scan{Scan: request}
		command := &pb.Command{Version: 1, Operation: variant}
		stream := testutil.OpenEvents(ctx, client, "records")
		if err := stream.Send(command); err != nil {
			t.Fatal(err)
		}
		count := 0
		var terminal *pb.ScanEnd
		for {
			event, err := stream.Recv()
			if err == io.EOF {
				break
			}
			if err != nil {
				t.Fatal("diagnostic Scan RPC failed", err)
			}
			if terminal != nil {
				t.Fatal("diagnostic Scan emitted after terminal result")
			}
			if document := event.GetDocument(); document != nil {
				var id, pad string
				var n int32
				if backend == "mongodb" {
					raw := bson.Raw(document.Data)
					id = raw.Lookup("_id").StringValue()
					pad = raw.Lookup("pad").StringValue()
					n = raw.Lookup("n").Int32()
				} else {
					var hit struct {
						ID     string `json:"_id"`
						Source struct {
							N   int32  `json:"n"`
							Pad string `json:"pad"`
						} `json:"_source"`
					}
					if json.Unmarshal(document.Data, &hit) != nil {
						t.Fatal("invalid diagnostic Search hit")
					}
					id, pad, n = hit.ID, hit.Source.Pad, hit.Source.N
				}
				if id != fmt.Sprintf("record_%05d", n) || n < 0 || n >= scanDiagnosticDocuments || pad != expectedPadding || seen[id] {
					t.Fatal("diagnostic Scan changed, omitted or repeated native documents", id)
				}
				seen[id] = true
				count++
			} else {
				terminal = event.GetScanEnd()
				if terminal == nil {
					t.Fatal("diagnostic Scan emitted an invalid event")
				}
			}
		}
		if terminal == nil || terminal.Failure != nil || terminal.DocumentCount != uint64(count) || count > 256 || terminal.Exhausted == (len(terminal.NextContinuationToken) != 0) {
			t.Fatal("diagnostic Scan terminal is invalid", terminal)
		}
		if terminal.Exhausted {
			if len(seen) != scanDiagnosticDocuments {
				t.Fatal("diagnostic Scan did not return every fixture document", len(seen))
			}
			return rpc
		}
		token = bytes.Clone(terminal.NextContinuationToken)
	}
	t.Fatal("diagnostic Scan did not exhaust")
	return 0
}

func TestScanDiagnosticMeasure(t *testing.T) {
	if os.Getenv("WEIR_SCAN_DIAGNOSTIC_MODE") != "measure" {
		t.Skip("diagnostic child only")
	}
	counts := newScanDiagnosticCounts()
	backend, target := os.Getenv("WEIR_SCAN_DIAGNOSTIC_BACKEND"), os.Getenv("WEIR_SCAN_DIAGNOSTIC_TARGET")
	var adapter execution.Adapter
	var err error
	if backend == "mongodb" {
		fixture := &testmongo.Fixture{URI: os.Getenv("WEIR_SCAN_DIAGNOSTIC_ENDPOINT")}
		proxy := testmongo.StartProxy(t, fixture)
		monitor := &event.CommandMonitor{}
		monitor.Started = func(_ context.Context, event *event.CommandStartedEvent) {
			size := 0
			if event.CommandName == "find" {
				size = int(event.Command.Lookup("limit").AsInt64())
			}
			counts.record(event.CommandName, size)
		}
		proxy.Monitor = monitor
		config := mongodb.Config{Store: "records", URI: proxy.URI(), Pool: 2, MaxReadSize: protocol.MaxDocument}
		adapter, err = mongodb.Open(t.Context(), config)
	} else {
		proxy := scanDiagnosticSearchProxy(t, os.Getenv("WEIR_SCAN_DIAGNOSTIC_ENDPOINT"), counts)
		config := search.Config{Store: "records", URL: proxy, Pool: 2, MaxReadSize: protocol.MaxDocument}
		adapter, err = search.Open(t.Context(), config)
	}
	if err != nil {
		t.Fatal(err)
	}
	limits := store.DefaultLimits()
	limits.Concurrency = 1
	limits.BackendTimeout = 8 * time.Second
	ingress := DefaultLimits()
	ingress.Stall = 60 * time.Second
	opts := routeAcceptanceNodeOptions{adapter: adapter, store: limits, limits: ingress}
	node := startRouteAcceptanceNode(t, opts)
	client := routeAcceptanceClient(t, node.address)
	scanDiagnosticTraverse(t, client, target, backend)
	assertRouteAcceptanceIdle(t, []*routeAcceptanceNode{node})
	for round := 1; round <= 3; round++ {
		before := counts.snapshot()
		started := time.Now()
		rpcs := scanDiagnosticTraverse(t, client, target, backend)
		elapsed := time.Since(started)
		after := counts.snapshot()
		assertRouteAcceptanceIdle(t, []*routeAcceptanceNode{node})
		result := &scanDiagnosticRound{Variant: os.Getenv("WEIR_SCAN_DIAGNOSTIC_VARIANT"), Source: os.Getenv("WEIR_SCAN_DIAGNOSTIC_SOURCE"), Backend: backend, Order: os.Getenv("WEIR_SCAN_DIAGNOSTIC_ORDER"), Round: round, Documents: scanDiagnosticDocuments, RPCs: rpcs, ElapsedNS: elapsed.Nanoseconds(), DocsSecond: scanDiagnosticDocuments / elapsed.Seconds(), Counts: scanDiagnosticDifference(after, before)}
		raw, err := json.Marshal(result)
		if err != nil {
			t.Fatal(err)
		}
		fmt.Printf("WEIR_SCAN_DIAGNOSTIC %s\n", raw)
	}
}

func TestScanDiagnosticController(t *testing.T) {
	if os.Getenv("WEIR_SCAN_DIAGNOSTIC_MODE") != "controller" {
		t.Skip("diagnostic controller only")
	}
	mongo := testmongo.Open(t)
	documents := make([]any, scanDiagnosticDocuments)
	for i := range documents {
		document := bson.D{{Key: "_id", Value: fmt.Sprintf("record_%05d", i)}, {Key: "n", Value: int32(i)}, {Key: "pad", Value: strings.Repeat("x", scanDiagnosticPayload)}}
		documents[i] = document
	}
	if _, err := mongo.Admin.Database(mongo.DB).Collection("records").InsertMany(t.Context(), documents); err != nil {
		t.Fatal(err)
	}
	search := testsearch.Open(t)
	var body strings.Builder
	for i := 0; i < scanDiagnosticDocuments; i++ {
		fmt.Fprintf(&body, "{\"index\":{\"_index\":%q,\"_id\":%q}}\n{\"n\":%d,\"pad\":%q}\n", search.Index, fmt.Sprintf("record_%05d", i), i, strings.Repeat("x", scanDiagnosticPayload))
	}
	code, raw := search.Do(t, "POST", "/_bulk?refresh=true", body.String())
	var reply struct {
		Errors bool `json:"errors"`
	}
	if code != http.StatusOK || json.Unmarshal(raw, &reply) != nil || reply.Errors {
		t.Fatal("diagnostic fixture seeding failed", code)
	}
	for _, backend := range []string{"mongodb", "search"} {
		endpoint, target := mongo.URI, mongo.DB+"/records"
		if backend == "search" {
			endpoint, target = search.URL, search.Index
		}
		for position, variant := range []string{"baseline", "current", "current", "baseline"} {
			binary := os.Getenv("WEIR_SCAN_DIAGNOSTIC_" + strings.ToUpper(variant) + "_BINARY")
			source := os.Getenv("WEIR_SCAN_DIAGNOSTIC_" + strings.ToUpper(variant) + "_SOURCE")
			command := exec.CommandContext(t.Context(), binary, "-test.run=^TestScanDiagnosticMeasure$", "-test.v", "-test.count=1", "-test.timeout=5m")
			command.Env = []string{"GOMAXPROCS=4", "WEIR_SCAN_DIAGNOSTIC_MODE=measure", "WEIR_SCAN_DIAGNOSTIC_BACKEND=" + backend, "WEIR_SCAN_DIAGNOSTIC_TARGET=" + target, "WEIR_SCAN_DIAGNOSTIC_ENDPOINT=" + endpoint, "WEIR_SCAN_DIAGNOSTIC_VARIANT=" + variant, "WEIR_SCAN_DIAGNOSTIC_SOURCE=" + source, fmt.Sprintf("WEIR_SCAN_DIAGNOSTIC_ORDER=%d", position+1)}
			command.Stdout, command.Stderr = os.Stdout, os.Stderr
			if err := command.Run(); err != nil {
				t.Fatal("diagnostic child failed", backend, variant, err)
			}
		}
	}
}
