package mongodb

import (
    "context"
    "testing"

    pb "github.com/batchstream/weir-protocol/api/weir/v1"
    "github.com/batchstream/weir/internal/execution"
    "go.mongodb.org/mongo-driver/v2/bson"
    "go.mongodb.org/mongo-driver/v2/event"
    "go.mongodb.org/mongo-driver/v2/x/mongo/driver/drivertest"
)

func TestReviewCrossNamespaceAuthorizationIsolation(t *testing.T) {
    for _, combined := range []bool{false, true} {
        name := "separate"
        if combined { name = "combined" }
        t.Run(name, func(t *testing.T) {
            var adapter *Adapter
            calls := 0
            monitor := &event.CommandMonitor{Started: func(_ context.Context, e *event.CommandStartedEvent) {
                deployment := adapter.deployment.(*drivertest.MockDeployment)
                if e.CommandName == "listCollections" {
                    collection := e.Command.Lookup("filter").Document().Lookup("name").StringValue()
                    deployment.AddResponses(collectionQualificationResponse("db", collection))
                    return
                }
                if e.CommandName == "endSessions" {
                    response := bson.D{{Key:"ok", Value:1}}
                    deployment.AddResponses(response)
                    return
                }
                if e.CommandName != "bulkWrite" { t.Fatal("unexpected command", e.CommandName) }
                calls++
                namespaces, _ := e.Command.Lookup("nsInfo").Array().Values()
                for _, ns := range namespaces {
                    if ns.Document().Lookup("ns").StringValue() == "db.forbidden" {
                        rejection := bson.D{{Key:"ok", Value:0}, {Key:"code", Value:13}, {Key:"errmsg", Value:"unauthorized"}}
                        deployment.AddResponses(rejection)
                        return
                    }
                }
                item := bson.D{{Key:"ok", Value:1}, {Key:"idx", Value:0}, {Key:"n", Value:1}}
                cursor := bson.D{{Key:"id", Value:int64(0)}, {Key:"ns", Value:"admin.$cmd.bulkWrite"}, {Key:"firstBatch", Value:bson.A{item}}}
                response := bson.D{{Key:"ok", Value:1}, {Key:"cursor", Value:cursor}, {Key:"nErrors", Value:0}, {Key:"nInserted", Value:0}, {Key:"nDeleted", Value:1}, {Key:"nMatched", Value:0}, {Key:"nModified", Value:0}, {Key:"nUpserted", Value:0}}
                deployment.AddResponses(response)
            }}
            adapter = batchMockAdapter(t, nil, monitor)
            var works []*execution.Plan
            for _, collection := range []string{"allowed", "forbidden"} {
                opts := batchOperationOptions{resource:"db/"+collection+"/s:item", action:"delete"}
                work, failure := prepareTestRecord(adapter, batchOperation(t, opts))
                if failure != nil { t.Fatal(failure) }
                works = append(works, work)
            }
            var results []*pb.Event
            if combined {
                results = adapter.executeRecords(t.Context(), works)
            } else {
                for _, work := range works {
                    group := []*execution.Plan{work}
                    results = append(results, adapter.executeRecords(t.Context(), group)...)
                }
            }
            t.Logf("bulk calls=%d allowed=%v forbidden=%v", calls, results[0].GetMutationResult(), results[1].GetMutationResult())
            if results[0].GetMutationResult().GetOutcome() != pb.MutationOutcome_APPLIED {
                t.Fatal("healthy namespace inherited another namespace authorization failure")
            }
            if results[1].GetMutationResult().GetFailure().GetCode() != pb.FailureCode_PERMISSION_DENIED {
                t.Fatal("forbidden namespace lost permission failure")
            }
        })
    }
}
