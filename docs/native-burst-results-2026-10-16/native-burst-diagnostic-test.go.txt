func TestNativeBurstDiagnostic(t *testing.T) {
	if os.Getenv("WEIR_NATIVE_BURST_DIAGNOSTIC") != "1" {
		t.Skip("explicit bounded diagnostic only; never qualification evidence")
	}
	binary := os.Getenv("WEIR_QUALIFICATION_BINARY")
	raw, err := os.ReadFile(binary)
	if err != nil {
		t.Fatal(err)
	}
	t.Logf("DIAGNOSTIC NONQUALIFICATION binary_SHA256=%x mode=%s", sha256.Sum256(raw), os.Getenv("WEIR_NATIVE_BURST_MODE"))
	mongo, search := testmongo.Open(t), testsearch.Open(t)
	metadata := strings.Fields(packagedDocker(t, "inspect", "--type", "container", "--format", `{{.Id}} {{index .Config.Labels "weir.owner"}} {{.Config.Image}} {{(index .NetworkSettings.Networks "bridge").IPAddress}} {{.State.Running}}`, "weir-local-elasticsearch"))
	if len(metadata) != 5 || len(metadata[0]) != 64 || metadata[1] != "weir-search-local" || metadata[2] != "docker.elastic.co/elasticsearch/elasticsearch@sha256:c2a3ed5f968be6d59c960aa0c60cfdaee667b6bc8211142021a41d0e85b43237" || metadata[4] != "true" {
		t.Fatal("owned pinned Search fixture identity mismatch", metadata)
	}
	bridge, err := netip.ParseAddr(metadata[3])
	if err != nil || !bridge.IsPrivate() || !bridge.Is4() {
		t.Fatal("owned Search bridge identity missing")
	}
	pads := []string{strings.Repeat("x", 1<<10), strings.Repeat("x", 256<<10), strings.Repeat("x", 2<<20), strings.Repeat("x", 8<<20)}
	datasets := []*qualificationDataset{
		{store: "mongo", root: mongo.DB + "/records", media: "application/bson", mongo: mongo, pads: pads, expected: make(map[string]qualificationRecord)},
		{store: "search", root: search.Index, media: "application/json", search: search, pads: pads, expected: make(map[string]qualificationRecord)},
	}
	for _, dataset := range datasets {
		seedQualificationDataset(t, dataset)
	}
	mongoBackend := &Mongo{URI: "mongodb://host.docker.internal:27028/?directConnection=true&serverMonitoringMode=poll"}
	searchBackend := &Search{URL: "http://" + bridge.String() + ":9200"}
	mongoLocal, searchLocal := &Local{Backend: BackendConfig{MongoDB: mongoBackend}}, &Local{Backend: BackendConfig{Search: searchBackend}}
	cfg := DefaultConfig()
	cfg.Routing.Stores = []StoreConfig{{Name: "mongo", Local: mongoLocal}, {Name: "search", Local: searchLocal}}
	opts := memoryRecoveryOptions{binary: binary, capacityMiB: 1024, automatic: true, config: &cfg}
	process := startMemoryRecoveryCLI(t, opts)
	t.Logf("DIAGNOSTIC NONQUALIFICATION runtime_identity=%s container=%s", strings.TrimSpace(packagedDocker(t, "exec", process.name, "/weir", "version")), process.name)
	initial := process.sample(t)
	_ = qualificationNativeMemory(t, initial, process.capacity)
	openContext, stop := context.WithTimeout(t.Context(), 3*time.Second)
	openOptions := weirclient.OpenOptions{Seed: process.address, Stores: []string{"mongo", "search"}}
	client, err := weirclient.Open(openContext, openOptions)
	stop()
	if err != nil {
		t.Fatal(err)
	}
	t.Cleanup(func() { _ = client.Close() })
	failures := 0
	for round := range 10 {
		burstOptions := nativeDiagnosticBurstOptions{process: process, client: client, datasets: datasets, round: round}
		failures += nativeDiagnosticBurstRun(t, burstOptions)
		for _, dataset := range datasets {
			verifyQualificationOracle(t, dataset)
		}
	}
	if err := client.Close(); err != nil {
		t.Fatal(err)
	}
	deadline := time.Now().Add(5 * time.Second)
	for {
		metrics := process.sample(t)
		native := qualificationNativeMemory(t, metrics, process.capacity)
		idle := true
		for _, name := range []string{"used_bytes", "resident_bytes", "leases", "starting", "settled", "borrowed", "waiting", "terminal"} {
			idle = idle && native[name] == 0
		}
		for _, name := range []string{"weir_store_pending_entries", "weir_store_pending_reserved_bytes", "weir_store_active_executions", "weir_store_paused_native_outputs", "weir_store_paused_native_reserved_bytes", "weir_store_result_reserved_entries", "weir_store_result_reserved_bytes", "weir_store_retained_result_reserved_bytes", "weir_store_working_reserved_bytes", "weir_store_publishers", "weir_ingress_sessions", "weir_ingress_queued_response_bytes"} {
			idle = idle && metrics[name] != nil && testmetrics.Sum(metrics, name) == 0
		}
		if idle {
			t.Logf("DIAGNOSTIC final ownership all0 within5s, Native=%v", native)
			break
		}
		if time.Now().After(deadline) {
			t.Fatal("DIAGNOSTIC fixed5s ownership cleanup failed", native)
		}
		time.Sleep(10 * time.Millisecond)
	}
	t.Logf("DIAGNOSTIC all_results offered=2560 received=2560 failures=%d no_retry=true no_replay=true", failures)
	if failures != 0 {
		t.Errorf("DIAGNOSTIC original fixed burst failed %d operations; full results and oracles retained", failures)
	}
}

type nativeDiagnosticBurstOptions struct {
	process  *memoryRecoveryCLI
	client   *weirclient.Client
	datasets []*qualificationDataset
	round    int
}

func nativeDiagnosticBurstRun(t *testing.T, opts nativeDiagnosticBurstOptions) int {
	t.Helper()
	ctx, cancel := context.WithTimeout(t.Context(), 5*time.Second)
	var pending sync.WaitGroup
	defer func() { cancel(); pending.Wait() }()
	results, tokens := make(chan qualificationResult, 512), make(chan struct{}, 512)
	var inFlight atomic.Int64
	work := qualificationWorkload{process: opts.process, client: opts.client, datasets: opts.datasets, ctx: ctx, results: results, tokens: tokens, pending: &pending, inFlight: &inFlight}
	started := time.Now().Add(100 * time.Millisecond)
	ticker := time.NewTicker(10 * time.Millisecond)
	defer ticker.Stop()
	launched, received, failures := false, 0, 0
	var finished time.Time
	stats := make(map[string]*qualificationStats)
	for received < 256 || time.Since(finished) < 100*time.Millisecond {
		select {
		case <-ctx.Done():
			t.Fatal("DIAGNOSTIC bounded round failed to collect all results", opts.round, received, ctx.Err())
		case scheduled := <-ticker.C:
			metrics := opts.process.sample(t)
			native := qualificationNativeMemory(t, metrics, opts.process.capacity)
			t.Logf("DIAGNOSTIC snapshot round=%d relative_us=%d sample_lag_us=%d Native=%v RSS=%.0f WS=%.0f latch=%.0f active=%.0f pending=%.0f", opts.round, time.Since(started).Microseconds(), time.Since(scheduled).Microseconds(), native, testmetrics.Sum(metrics, "process_resident_memory_bytes"), testmetrics.Sum(metrics, "weir_memory_cgroup_working_set_bytes"), testmetrics.Sum(metrics, "weir_memory_latched"), testmetrics.Sum(metrics, "weir_store_active_executions"), testmetrics.Sum(metrics, "weir_store_pending_entries"))
			if !launched && !time.Now().Before(started) {
				launched = true
				for index := range 256 {
					job := qualificationJob{sequence: 126000 + opts.round*256 + index, backend: index % 2, kind: index % 3, scheduled: started, phase: "burst"}
					launchQualificationJob(work, job)
				}
			}
		case result := <-results:
			received++
			job := result.job
			key := opts.datasets[job.backend].store + "/" + qualificationMethod(job.kind)
			appendQualificationStats(stats, key, result)
			if result.err == nil {
				rememberQualificationMutation(opts.datasets[job.backend], job)
			} else {
				failures++
			}
			t.Logf("DIAGNOSTIC result round=%d seq=%d backend=%s method=%s e2e=%s service=%s lag=%s error=%v", opts.round, job.sequence, opts.datasets[job.backend].store, qualificationMethod(job.kind), result.latency, result.service, result.lag, result.err)
			if received == 256 {
				finished = time.Now()
			}
		}
	}
	pending.Wait()
	reportQualificationStats(t, stats, false)
	t.Logf("DIAGNOSTIC burst round=%d offered=256 received=%d failures=%d", opts.round, received, failures)
	return failures
}
