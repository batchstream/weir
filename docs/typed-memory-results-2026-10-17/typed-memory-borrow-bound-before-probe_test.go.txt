package store

import (
    "testing"

    "github.com/batchstream/weir-protocol/api/protocol"
    "github.com/batchstream/weir/internal/execution"
)

// The fixture describes already-owned typed results and an in-flight Native
// reply using ledger calls only. It allocates no document buffers and has no
// backend. It isolates the protected-IO contract, not a live rejection instant.
func TestReview81DesiredShortNativeProgressAfterLargeNativeBorrows(t *testing.T) {
    memory := testExecutionMemory(750_000_000)
    nativePlan := plan(1, "large-native", false)
    nativePlan.Command = nativeCall()
    large, failure := memory.admit(nativePlan)
    if failure != nil { t.Fatal(failure) }
    defer large.release()
    started, waiting := large.start(48_008_256, defaultNativeIOReserve)
    if !started || waiting { t.Fatal("initial Native peak did not fit", memory.Snapshot()) }
    var typed []*executionLease
    defer func() { for _, lease := range typed { lease.release() } }()
    for index := range 13 {
        work := plan(uint64(index+2), "retained-typed", true)
        lease, failure := memory.admit(work)
        if failure != nil { t.Fatal(failure) }
        typed = append(typed, lease)
        started, waiting = lease.start(0, 0)
        if !started || waiting { t.Fatal("result producer did not fit", memory.Snapshot()) }
        capacity := 16 << 20
        if index == 12 { capacity = 7 << 20 }
        if err := lease.reserveResult(capacity + execution.ResultOverheadBytes); err != nil { t.Fatal("retained result did not fit", err, memory.Snapshot()) }
        lease.finishWorking()
        snapshot := memory.Snapshot()
        if snapshot.UsedBytes > snapshot.BudgetBytes || snapshot.ResidentBytes > snapshot.BudgetBytes-snapshot.IOReserveBytes {
            t.Fatal("fixture crossed an existing budget", snapshot)
        }
    }
    before := memory.Snapshot()
    if err := large.settle((12<<20)+execution.EntryOverheadBytes, protocol.NativeChunk+20); err != nil { t.Fatal("already-reserved actual reply could not settle", err) }
    settled := memory.Snapshot()
    borrowedBytes := settled.UsedBytes-settled.ResidentBytes
    if settled.Borrowed != 1 || settled.Starting != 0 || borrowedBytes <= executionBorrowWindow || settled.UsedBytes > settled.BudgetBytes {
        t.Fatal("fixture did not isolate a whole large borrowed response", settled)
    }
    shortPlan := plan(100, "healthy-short-native", false)
    shortPlan.Command = nativeCall()
    short, failure := memory.admit(shortPlan)
    if failure != nil { t.Fatal("small input could not enter existing queue promise", failure) }
    defer short.release()
    const shortWorking = 48_008_448
    started, waiting = short.start(shortWorking, defaultNativeIOReserve)
    after := memory.Snapshot()
    t.Logf("before=%+v settled=%+v borrowed_bytes=%d protected_reserve=%d free_after_settle=%d short_working=%d short_publication=%d after=%+v", before, settled, borrowedBytes, settled.IOReserveBytes, settled.BudgetBytes-settled.UsedBytes, shortWorking, short.publication-nativeQueuedWindow, after)
    if !started || waiting {
        t.Fatal("healthy short Native cannot use the promised protected IO reserve after a large paused Native reply borrowed it")
    }
}
