// DIAGNOSTIC ONLY: this bounded burst is not the sustained qualification gate.
// Inject this ignored file through its accompanying Go overlay after the owner
// explicitly opens the isolated DB/CPU window. It does not alter CLI production.
func TestDiagnosticCLITypedLargeReadBurst(t *testing.T) {
	if os.Getenv("WEIR_TYPED_LARGE_DIAGNOSTIC") != "1" {
		t.Skip("explicit isolated typed-large-read diagnostic window required")
	}
	if os.Getenv("WEIR_MONGO_PROFILE") == "tls" {
		t.Fatal("this diagnostic uses only the independently checked plain local fixture")
	}
	receiptPath := os.Getenv("WEIR_TYPED_LARGE_RECEIPT")
	output := os.Getenv("WEIR_TYPED_LARGE_OUTPUT_DIR")
	if !filepath.IsAbs(receiptPath) || !filepath.IsAbs(output) {
		t.Fatal("explicit absolute receipt and ignored output directory required")
	}
	raw, err := os.ReadFile(receiptPath)
	if err != nil {
		t.Fatal(err)
	}
	var receipt typedLargeReceipt
	if err := json.Unmarshal(raw, &receipt); err != nil {
		t.Fatal(err)
	}
	if receipt.Source != "fde3b1775b0a36dbb4f4044be1299f731db43314" || receipt.State != "clean-commit" || receipt.Target != "linux/arm64" || receipt.SHA256 != "795601c60e8f4398a8108843939361032f88249acc8f75ee42ae322ca7269c79" || !filepath.IsAbs(receipt.Binary) {
		t.Fatal("C0 receipt differs from the explicitly frozen production binary", receipt)
	}
	binary, err := os.ReadFile(receipt.Binary)
	if err != nil {
		t.Fatal(err)
	}
	if fmt.Sprintf("%x", sha256.Sum256(binary)) != receipt.SHA256 {
		t.Fatal("C0 binary bytes differ from the receipt")
	}
	binary = nil
	if err := os.MkdirAll(output, 0755); err != nil {
		t.Fatal(err)
	}
	fixture := testmongo.Open(t)
	pad := strings.Repeat("x", 12<<20)
	requests := make([]typedLargeRequest, 32)
	for index := range requests {
		id := fmt.Sprintf("typed-large-%02d", index)
		document := bson.D{{Key: "_id", Value: id}, {Key: "n", Value: index}, {Key: "pad", Value: pad}}
		encoded, err := bson.Marshal(document)
		if err != nil {
			t.Fatal(err)
		}
		ctx, cancel := context.WithTimeout(t.Context(), 5*time.Second)
		_, err = fixture.Admin.Database(fixture.DB).Collection("records").InsertOne(ctx, document)
		cancel()
		if err != nil {
			t.Fatal("owned document seed failed", id, err)
		}
		requests[index] = typedLargeRequest{ID: id, Resource: fixture.DB + "/records/s:" + id, Bytes: len(encoded), SHA256: fmt.Sprintf("%x", sha256.Sum256(encoded))}
	}
	for _, capacity := range []int{512, 1024} {
		t.Run(fmt.Sprintf("%dMiB", capacity), func(t *testing.T) {
			backend := &Mongo{URI: "mongodb://host.docker.internal:27028/?directConnection=true&serverMonitoringMode=poll"}
			connection := BackendConfig{MongoDB: backend}
			local := &Local{Backend: connection}
			definition := StoreConfig{Name: "mongo", Local: local}
			cfg := DefaultConfig()
			cfg.Routing.Stores = []StoreConfig{definition}
			if cfg.Basic.Transport.MaxPendingRecords != 32 {
				t.Fatal("diagnostic changed default per-stream admission window")
			}
			opts := memoryRecoveryOptions{binary: receipt.Binary, capacityMiB: capacity, automatic: true, config: &cfg}
			service := startMemoryRecoveryCLI(t, opts)
			evidence := typedLargeEvidence{Diagnostic: true, Receipt: receipt, CapacityMiB: capacity, Store: "mongo", Requests: append([]typedLargeRequest(nil), requests...), StartedAt: time.Now().UTC(), StreamWindow: 64 << 10, ConnectionWindow: 4 << 20, NoRecvMilliseconds: 3000, CallerBudgetMilliseconds: 5000}
			filename := filepath.Join(output, fmt.Sprintf("typed-large-C0-%dMiB-%d.json", capacity, time.Now().UnixNano()))
			// LIFO cleanup records state before the shared owned-CLI helper stops
			// and removes its container. OOM is expected diagnostic failure evidence.
			t.Cleanup(func() {
				evidence.FinalState = typedLargeInspect(service.name)
				data, err := json.MarshalIndent(evidence, "", "  ")
				if err != nil {
					t.Error("diagnostic evidence encoding failed", err)
					return
				}
				if err := os.WriteFile(filename, append(data, '\n'), 0644); err != nil {
					t.Error("diagnostic evidence write failed", err)
					return
				}
				t.Logf("DIAGNOSTIC ONLY evidence=%s OOMKilled=%t running=%t exit=%d successful_ID_matches=%d", filename, evidence.FinalState.OOMKilled, evidence.FinalState.Running, evidence.FinalState.ExitCode, evidence.ValidatedDocuments)
			})
			initial := service.sample(t)
			if testmetrics.Sum(initial, "weir_memory_go_runtime_managed") != 1 || testmetrics.Sum(initial, "weir_store_active_executions_limit") != 16 || testmetrics.Sum(initial, "weir_store_pending_entries_limit") != 1024 || testmetrics.Sum(initial, "weir_store_pending_reserved_bytes_limit") != 32<<20 || testmetrics.Sum(initial, "weir_store_batch_max_operations") != 128 || testmetrics.Sum(initial, "weir_store_batch_max_bytes") != 4<<20 || testmetrics.Sum(initial, "weir_store_batch_max_wait_seconds") != .02 {
				t.Fatal("diagnostic changed default CLI memory/scheduler policy")
			}
			evidence.Version = strings.TrimSpace(packagedDocker(t, "exec", service.name, "/weir", "version"))
			var identity map[string]string
			if err := json.Unmarshal([]byte(evidence.Version), &identity); err != nil {
				t.Fatal("running CLI identity is invalid", err)
			}
			if identity["state"] != "clean-commit" || identity["dirty"] != "false" || identity["revision"] != receipt.Source || identity["target"] != receipt.Target {
				t.Fatal("running CLI is not the exact clean C0 binary", evidence.Version)
			}
			evidence.Before = typedLargeObserve(service)
			evidence.Before.MemoryEvents, evidence.Before.MemoryEventsError = typedLargeMemoryEvents(service.name)
			evidence.LastSuccessfulMetrics = evidence.Before.Metrics
			evidence.PeakObservedMetrics = make(map[string]float64)
			typedLargeBurst(t, service, &evidence)
		})
	}
}

func typedLargeBurst(t *testing.T, service *memoryRecoveryCLI, evidence *typedLargeEvidence) {
	t.Helper()
	dialOptions := []grpc.DialOption{grpc.WithTransportCredentials(insecure.NewCredentials()), grpc.WithNoProxy(), grpc.WithDisableRetry(), grpc.WithDisableServiceConfig(), grpc.WithStaticStreamWindowSize(64 << 10), grpc.WithStaticConnWindowSize(4 << 20), grpc.WithDefaultCallOptions(grpc.MaxCallRecvMsgSize(20 << 20))}
	connection, err := grpc.NewClient("passthrough:///"+service.address, dialOptions...)
	if err != nil {
		t.Fatal(err)
	}
	defer connection.Close()
	client := pb.NewStoreServiceClient(connection)
	ctx, cancel := context.WithTimeout(t.Context(), 5*time.Second)
	defer cancel()
	streams := make([]grpc.BidiStreamingClient[pb.ExecuteRequest, pb.ExecuteResponse], len(evidence.Requests))
	for index := range streams {
		streamContext := metadata.AppendToOutgoingContext(ctx, "weir-request-id", fmt.Sprintf("typed-large-diagnostic-%d-%02d", evidence.CapacityMiB, index))
		streams[index], err = client.Execute(streamContext)
		if err != nil {
			t.Fatal("diagnostic stream open failed before barrier", index, err)
		}
	}
	barrier := make(chan struct{})
	var senders sync.WaitGroup
	for index := range streams {
		senders.Go(func() {
			<-barrier
			read := &pb.ReadRequest{Resource: evidence.Requests[index].Resource}
			operation := &pb.Command_Read{Read: read}
			command := &pb.Command{Operation: operation}
			request := &pb.ExecuteRequest{StoreName: evidence.Store, Index: 1, Command: command}
			err := streams[index].Send(request)
			if err == nil {
				err = streams[index].CloseSend()
			}
			evidence.Requests[index].Sent = err == nil
			if err != nil {
				evidence.Requests[index].SendError = err.Error()
			}
		})
	}
	started := time.Now()
	close(barrier)
	monitorDeadline := started.Add(3 * time.Second)
	for time.Now().Before(monitorDeadline) {
		observation := typedLargeObserve(service)
		evidence.Observations = append(evidence.Observations, observation)
		if observation.Metrics != nil {
			evidence.LastSuccessfulMetrics = observation.Metrics
			for name, value := range observation.Metrics {
				evidence.PeakObservedMetrics[name] = max(evidence.PeakObservedMetrics[name], value)
			}
		}
		remaining := time.Until(monitorDeadline)
		if remaining > 0 {
			time.Sleep(min(remaining, 100*time.Millisecond))
		}
	}
	evidence.NoRecvActualMilliseconds = time.Since(started).Milliseconds()
	senders.Wait()
	for _, request := range evidence.Requests {
		if request.Sent {
			evidence.SentRequests++
		}
	}
	// Only after the fixed hold ends may clients consume; each goroutine drops
	// its document after validation. The original common five-second deadline
	// bounds all response consumption, even when the service has died.
	var receivers sync.WaitGroup
	for index := range streams {
		receivers.Go(func() {
			response, err := streams[index].Recv()
			record := &evidence.Requests[index]
			record.ReceiveStatus = status.Code(err).String()
			if err != nil {
				record.ReceiveError = err.Error()
				return
			}
			read := response.GetEvent().GetReadResult()
			if read.GetFailure() != nil {
				record.ReceiveStatus = "TYPED_READ_FAILURE"
				record.FailureCode = read.GetFailure().Code.String()
				record.ReceiveError = read.GetFailure().Message
				return
			}
			if response.Index != 1 || read == nil || read.GetDocument() == nil || read.GetDocument().ContentType != "application/bson" {
				record.ReceiveStatus = codes.Internal.String()
				record.ReceiveError = fmt.Sprintf("missing typed document or mismatched index; failure=%v", read.GetFailure())
				return
			}
			document := read.GetDocument().Data
			record.DocumentID, _ = bson.Raw(document).Lookup("_id").StringValueOK()
			record.DocumentBytes = len(document)
			record.DocumentSHA256 = fmt.Sprintf("%x", sha256.Sum256(document))
			record.IDAndHashMatch = record.DocumentID == record.ID && record.DocumentBytes == record.Bytes && record.DocumentSHA256 == record.SHA256
			if !record.IDAndHashMatch {
				record.ReceiveStatus = codes.Internal.String()
				record.ReceiveError = "document ID/BSON bytes/hash differs from the seeded request"
			}
		})
	}
	receivers.Wait()
	cancel()
	_ = connection.Close()
	for _, request := range evidence.Requests {
		if request.IDAndHashMatch {
			evidence.ValidatedDocuments++
		}
		if request.DocumentBytes > 0 && !request.IDAndHashMatch {
			t.Error("typed Read returned a document that differs from the seeded ID/hash", request.ID, request.DocumentID, request.DocumentBytes)
		}
	}
	evidence.AfterCancel = typedLargeObserve(service)
	evidence.AfterCancel.MemoryEvents, evidence.AfterCancel.MemoryEventsError = typedLargeMemoryEvents(service.name)
	if evidence.AfterCancel.Metrics != nil {
		evidence.LastSuccessfulMetrics = evidence.AfterCancel.Metrics
	}
	t.Logf("DIAGNOSTIC ONLY cap=%dMiB sent=%d expected=32 hold=%dms validated=%d final_running=%t final_OOMKilled=%t last_wire=%.0f last_results=%.0f", evidence.CapacityMiB, evidence.SentRequests, evidence.NoRecvActualMilliseconds, evidence.ValidatedDocuments, evidence.AfterCancel.State.Running, evidence.AfterCancel.State.OOMKilled, evidence.AfterCancel.Metrics["weir_ingress_queued_response_bytes"], evidence.AfterCancel.Metrics["weir_store_result_reserved_bytes"])
}

func typedLargeObserve(service *memoryRecoveryCLI) typedLargeObservation {
	observation := typedLargeObservation{At: time.Now().UTC()}
	var workers sync.WaitGroup
	workers.Go(func() {
		client := &http.Client{Timeout: 150 * time.Millisecond}
		response, err := client.Get("http://" + service.diagnostic + "/metrics")
		if err != nil {
			observation.MetricsError = err.Error()
			return
		}
		defer response.Body.Close()
		if response.StatusCode != http.StatusOK {
			observation.MetricsError = fmt.Sprintf("HTTP %d", response.StatusCode)
			return
		}
		parser := expfmt.NewTextParser(model.LegacyValidation)
		families, err := parser.TextToMetricFamilies(io.LimitReader(response.Body, 4<<20))
		if err != nil {
			observation.MetricsError = err.Error()
			return
		}
		observation.Metrics = make(map[string]float64)
		for _, name := range []string{"process_resident_memory_bytes", "go_memstats_heap_alloc_bytes", "go_memstats_heap_idle_bytes", "go_memstats_heap_released_bytes", "weir_memory_cgroup_current_bytes", "weir_memory_cgroup_working_set_bytes", "weir_memory_latched", "weir_memory_go_runtime_limit_bytes", "weir_store_pending_entries", "weir_store_active_executions", "weir_store_result_reserved_entries", "weir_store_result_reserved_bytes", "weir_store_retained_result_reserved_bytes", "weir_store_working_reserved_bytes", "weir_ingress_queued_response_bytes", "weir_ingress_sessions", "weir_native_memory_leases", "weir_native_memory_used_bytes"} {
			if families[name] != nil {
				observation.Metrics[name] = testmetrics.Sum(families, name)
			}
		}
	})
	workers.Go(func() { observation.State = typedLargeInspect(service.name) })
	workers.Wait()
	return observation
}

func typedLargeMemoryEvents(name string) (string, string) {
	ctx, cancel := context.WithTimeout(context.Background(), 300*time.Millisecond)
	defer cancel()
	raw, err := exec.CommandContext(ctx, "docker", "exec", name, "cat", "/sys/fs/cgroup/memory.events").CombinedOutput()
	if err != nil {
		return "", fmt.Sprintf("%v: %s", err, strings.TrimSpace(string(raw)))
	}
	return strings.TrimSpace(string(raw)), ""
}

func typedLargeInspect(name string) typedLargeState {
	state := typedLargeState{}
	ctx, cancel := context.WithTimeout(context.Background(), 300*time.Millisecond)
	defer cancel()
	format := `{"owner":{{json (index .Config.Labels "weir.owner")}},"running":{{json .State.Running}},"oom_killed":{{json .State.OOMKilled}},"exit_code":{{json .State.ExitCode}}}`
	raw, err := exec.CommandContext(ctx, "docker", "inspect", "--type", "container", "--format", format, name).CombinedOutput()
	if err != nil {
		state.Error = fmt.Sprintf("%v: %s", err, strings.TrimSpace(string(raw)))
		return state
	}
	if err := json.Unmarshal(raw, &state); err != nil {
		state.Error = err.Error()
		return state
	}
	if state.Owner != name {
		state.Error = "container owner differs from the diagnostic owner"
	}
	return state
}
