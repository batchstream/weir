package main

import (
 "context"
 "encoding/json"
 "fmt"
 "io"
 "net"
 "net/http"
 "os"
 "slices"
 "sync"
 "time"

 "github.com/batchstream/weir-protocol/api/netlimit"
)

type observation struct { Mode string; Sequence int; Total, DNS, TCP, Headers time.Duration; Addresses []string; Stage, Error string; Status int }
type key struct{}
func probe(mode string, sequence int) observation {
 observation := observation{Mode:mode, Sequence:sequence}
 started := time.Now()
 ctx, cancel := context.WithTimeout(context.Background(), 3*time.Second)
 defer cancel()
 dial := func(ctx context.Context, network, address string) (net.Conn,error) {
  ctx, cancel := context.WithTimeout(ctx, 2*time.Second)
  defer cancel()
  host, port, err := net.SplitHostPort(address)
  if err != nil { return nil,err }
  if mode == "dns" {
   before := time.Now()
   ips, err := netlimit.LookupHost(ctx,nil,host)
   observation.DNS = time.Since(before)
   observation.Addresses = ips
   if err != nil { observation.Stage="dns"; return nil,err }
   host = ips[0]
  }
  before := time.Now()
  dialer := net.Dialer{Timeout:2*time.Second}
  conn,err := dialer.DialContext(ctx,network,net.JoinHostPort(host,port))
  observation.TCP = time.Since(before)
  if err != nil { observation.Stage="tcp" }
  return conn,err
 }
 protocols := &http.Protocols{}
 protocols.SetHTTP1(true)
 transport := &http.Transport{Proxy:nil,DialContext:dial,Protocols:protocols,DisableCompression:true,DisableKeepAlives:true}
 defer transport.CloseIdleConnections()
 client := &http.Client{Transport:transport}
 endpoint := "http://host.docker.internal:19200/"
 if mode=="ip" { endpoint="http://192.168.65.254:19200/" }; if mode=="direct" { endpoint="http://172.17.0.2:9200/" }
 request,err := http.NewRequestWithContext(ctx,http.MethodGet,endpoint,nil)
 if err != nil { panic(err) }
 response,err := client.Do(request)
 observation.Headers=time.Since(started)
 if err != nil { observation.Error=err.Error(); if observation.Stage=="" { observation.Stage="http_headers" } } else {
  observation.Status=response.StatusCode
  _,err=io.Copy(io.Discard,response.Body)
  closeErr:=response.Body.Close()
  if err==nil { err=closeErr }
  if err != nil { observation.Stage="body"; observation.Error=err.Error() }
 }
 observation.Total=time.Since(started)
 return observation
}

func main() {
 for _,name:=range []string{"/proc/net/snmp","/proc/net/netstat"} {raw,err:=os.ReadFile(name);fmt.Printf("NETSTAT_START %s %s %v\n",name,raw,err)}
 const duration=120*time.Second
 const ticks=1800
 results:=make(chan observation,ticks*2)
 var pending sync.WaitGroup
 start:=time.Now()
 for sequence:=range ticks {
  if remaining:=time.Until(start.Add(time.Duration(sequence)*duration/ticks)); remaining>0 { time.Sleep(remaining) }
  for _,mode:=range []string{"direct","ip"} {
   pending.Add(1)
   go func(){defer pending.Done();results<-probe(mode,sequence)}()
  }
 }
 pending.Wait();close(results)
 for _,name:=range []string{"/proc/net/snmp","/proc/net/netstat"} {raw,err:=os.ReadFile(name);fmt.Printf("NETSTAT_END %s %s %v\n",name,raw,err)}
 encoder:=json.NewEncoder(os.Stdout)
 groups:=make(map[string][]observation)
 for result:=range results {
  groups[result.Mode]=append(groups[result.Mode],result)
  if result.Total>25*time.Millisecond || result.Error!="" { _=encoder.Encode(result) }
 }
 for _,mode:=range []string{"direct","ip"} {
  values:=groups[mode]
  slices.SortFunc(values,func(a,b observation) int { if a.Total<b.Total{return -1};if a.Total>b.Total{return 1};return 0 })
  failures:=0
  for _,value:=range values {if value.Error!=""||value.Status!=200 {failures++}}
  fmt.Printf("SUMMARY mode=%s count=%d failures=%d p50=%s p99=%s max=%s\n",mode,len(values),failures,values[len(values)/2].Total,values[(len(values)*99+99)/100-1].Total,values[len(values)-1].Total)
 }
}
