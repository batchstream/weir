package main
import (
 "context"
 "encoding/json"
 "errors"
 "os"
 "runtime"
 "strings"
 "github.com/batchstream/weir/internal/execution"
 "github.com/batchstream/weir/internal/luaengine"
 "github.com/batchstream/weir/internal/value"
)
func main() {
 text := strings.Repeat("0", 1<<16)+"1.0"
 scalar := value.Value{Kind:value.String, Text:text}
 field := value.Field{Name:"n", Value:scalar}
 input := value.Value{Kind:value.Object, Fields:[]value.Field{field}}
 for _, source := range []string{
  `return function(_,input) local s=input.n; for i=1,128 do for j=s,0 do end end; return weir.keep() end`,
  `return function(_,input) local s=input.n; for i=1,128 do table.concat({},'',s,0) end; return weir.keep() end`,
 } {
  reserved:=0
  program:=luaengine.Program{Source:source, Input:input, Reserve:func(total int)error { if total > 1<<20 {return execution.ErrExecutionMemory};reserved=total;return nil }}
  var before,after runtime.MemStats
  runtime.GC()
  runtime.ReadMemStats(&before)
  result,err:=luaengine.Evaluate(context.Background(),program)
  runtime.ReadMemStats(&after)
  record:=map[string]any{"source":source,"inputBytes":len(text),"budgetLimit":1<<20,"reserved":reserved,"allocated":after.TotalAlloc-before.TotalAlloc,"action":result.Action,"memoryRefused":errors.Is(err,execution.ErrExecutionMemory)}
  if err!=nil {record["error"]=err.Error()}
  json.NewEncoder(os.Stdout).Encode(record)
 }
}
